cmake_minimum_required(VERSION 3.0.2)
project(ag_client)
find_package(catkin REQUIRED COMPONENTS rospy std_msgs slicer_air_ground_msgs)
catkin_python_setup()
catkin_package(CATKIN_DEPENDS rospy std_msgs slicer_air_ground_msgs)
catkin_install_python(PROGRAMS scripts/ag_client_cli.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY ../docs DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(DIRECTORY launch config DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
if(CATKIN_ENABLE_TESTING)
  catkin_add_nosetests(test/test_protocol.py)
  find_package(rostest REQUIRED)
  add_rostest(test/client.test)
endif()
