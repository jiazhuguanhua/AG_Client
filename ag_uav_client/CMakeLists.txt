cmake_minimum_required(VERSION 3.0.2)
project(ag_uav_client)
find_package(catkin REQUIRED COMPONENTS roscpp rospy std_msgs slicer_air_ground_msgs)
find_package(PkgConfig REQUIRED)
pkg_check_modules(JSONCPP REQUIRED jsoncpp)
find_package(OpenSSL REQUIRED)
catkin_python_setup()
catkin_package(INCLUDE_DIRS include LIBRARIES ag_client_cpp CATKIN_DEPENDS roscpp rospy std_msgs slicer_air_ground_msgs)
add_library(ag_client_cpp src/cpp/client.cpp)
target_compile_features(ag_client_cpp PUBLIC cxx_std_14)
target_include_directories(ag_client_cpp PUBLIC include ${catkin_INCLUDE_DIRS} PRIVATE ${JSONCPP_INCLUDE_DIRS} ${OPENSSL_INCLUDE_DIR})
target_link_libraries(ag_client_cpp ${catkin_LIBRARIES} ${JSONCPP_LIBRARIES} OpenSSL::Crypto)
add_dependencies(ag_client_cpp ${catkin_EXPORTED_TARGETS})
install(TARGETS ag_client_cpp ARCHIVE DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION} LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(DIRECTORY include/ag_client/ DESTINATION ${CATKIN_GLOBAL_INCLUDE_DESTINATION}/ag_client)
catkin_install_python(PROGRAMS scripts/test_menu.py
  scripts/ag_uav_gateway.py scripts/v2_send.py
  DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY ../docs DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(DIRECTORY launch config DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
if(CATKIN_ENABLE_TESTING)
  catkin_add_nosetests(test/test_protocol.py)
  catkin_add_nosetests(test/test_observation.py)
endif()
