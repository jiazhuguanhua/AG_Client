cmake_minimum_required(VERSION 3.0.2)
project(ag_ugv_task)
find_package(catkin REQUIRED COMPONENTS ag_client roslib rospy std_msgs)
catkin_python_setup()
catkin_package(CATKIN_DEPENDS ag_client roslib rospy std_msgs)
catkin_install_python(PROGRAMS scripts/ag_ugv_task_node.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY launch DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
if(CATKIN_ENABLE_TESTING)
  catkin_add_nosetests(test/test_route.py)
endif()
